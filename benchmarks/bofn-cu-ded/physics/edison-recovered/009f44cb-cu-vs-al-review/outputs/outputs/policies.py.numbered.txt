97: def recommend(net: Network, d: Data, truth: AlloyTruth, it: int):
98:     """True objective at argmax of the network posterior mean (paper's metric)."""
99:     g = torch.Generator().manual_seed(77 + it)
100:     pool = [sobol(1024, D, seed=it)]
101:     if d.full:
102:         pool.append(torch.stack([x for x, _ in d.full]))
103:     if d.builds:
104:         pool.append(torch.stack([design_of_build(d, j) for j in range(len(d.builds))]))
105:     P = torch.cat(pool)
106:     nu = net.mean(P, eps_for(P.shape[0], S_REC, 5))
107:     for _ in range(2):
108:         top = P[nu.topk(8).indices]
109:         Q = local_perturb(top, 32, 0.04, g)
110:         nq = net.mean(Q, eps_for(Q.shape[0], S_REC, 5))
111:         P, nu = torch.cat([P, Q]), torch.cat([nu, nq])
112:     xr = P[nu.argmax()]
113:     return float(truth.objective(xr.unsqueeze(0))[0]), xr

116: class Recorder:
117:     def __init__(self, truth, d):
118:         self.truth, self.d = truth, d
119:         self.rows = []
120: 
121:     def __call__(self, net, it, action):
122:         if REC:
123:             rec, xr = recommend(net, self.d, self.truth, it)
124:         else:
125:             rec, xr = float("nan"), torch.zeros(D)
126:         best = max(v for _, v in self.d.full) if self.d.full else float("nan")
127:         # best deposited alloy by the model (tested or not), scored by its true value
128:         Xb = torch.stack([design_of_build(self.d, j) for j in range(len(self.d.builds))])
129:         nub = net.mean(Xb, eps_for(Xb.shape[0], S_REC, 6))
130:         printed = float(self.truth.objective(Xb[nub.argmax()].unsqueeze(0))[0])
131:         self.rows.append(
132:             {"spent": self.d.spent, "rec": rec, "best_validated": best, "rec_printed": printed, "action": action, "x_rec": [round(v, 4) for v in xr.tolist()]}
133:         )

136: def w_gate(d: Data):
137:     """The one-hour screen's gate: the top quartile of every screened build so far (seed builds included)."""
138:     return float(torch.tensor(d.obs["w"][1]).quantile(W_QUANTILE))

144: def w_reference(truth: AlloyTruth):
145:     """One-hour screen index of the SCREEN_REF reference coupon deposited dense (its densest build over
146:     16,384 settings), as a lab would measure it alongside its builds; it passes immersion in every cell."""
147:     key = (truth.kappa, truth.eta)
148:     if key not in _W_REF:
149:         c = torch.tensor(bofn_alloy.REFERENCE[SCREEN_REF])
150:         apv = torch.quasirandom.SobolEngine(3, scramble=True, seed=0).draw(16384).double()
151:         o = truth.full(torch.cat([c.expand(apv.shape[0], 4), apv], -1))
152:         _W_REF[key] = float(o["w"][o["rho_m"].argmax()])
153:     return _W_REF[key]

161: # --------------------------------------------------------------------------- full-evaluation policies
162: def expected_improvement(gp, P, best):
163:     m, v = gp.post_std(P)
164:     mu, sd = m * gp.ys + gp.ym, v.sqrt() * gp.ys
165:     z = (mu - best) / sd
166:     nrm = torch.distributions.Normal(0.0, 1.0)
167:     return sd * (z * nrm.cdf(z) + nrm.log_prob(z).exp())

170: def calphad_shortlist(d, lab, seed, top=SCREEN_TOP, ni_floor=None):
171:     """Stage 1 of tiered screening: simulate SCREEN_N compositions, keep the `top`
172:     strongest that CALPHAD predicts crack-free (solidification index at or below
173:     F_SAFE), net of the critical-mineral penalty. On this network that shortlist is
174:     aluminum bronzes with 2 to 6 wt% Ni, which the 30-day immersion test fails.
175: 
176:     With `ni_floor` (corrosion-aware tiered screening, `tiered_ni`), only compositions with at
177:     least that much nickel are eligible. So few of them pass the crack screen (none in 14 of 20
178:     seeds at 10 wt%) that the rule needs a decision: when fewer than `top` pass, the shortlist is
179:     filled with the eligible compositions CALPHAD rates least crack-prone (lowest f), i.e. the
180:     crack screen becomes a ranking rather than a cutoff."""
181:     for c in sobol(SCREEN_N, NC, seed=4000 + seed)[: SCREEN_N - N_CALPHAD0]:
182:         lab.calphad(c)
183:     comps = torch.tensor([list(k) for k in d.calphad])
184:     s = torch.tensor([v["s"] for v in d.calphad.values()])
185:     f = torch.tensor([v["f"] for v in d.calphad.values()])
186:     fom = s - LAMBDA * burden(torch.cat([comps, torch.zeros(len(comps), D - NC)], -1))
187:     eligible = comps[:, INI] >= ni_floor - 1e-12 if ni_floor is not None else torch.ones(len(comps), dtype=torch.bool)
188:     fom = torch.where((f <= F_SAFE) & eligible, fom, torch.full_like(fom, -9.0))
189:     order = fom.argsort(descending=True)
190:     order = order[fom[order] > -9.0][:top]
191:     if ni_floor is not None and len(order) < top:
192:         rest = torch.where(eligible & (f > F_SAFE))[0]
193:         order = torch.cat([order, rest[f[rest].argsort()][: top - len(order)]])
194:     return comps[order], comps, fom, f

205: def chain(lab, d, lot, pv, need_utah, rows, w_ref=None):
206:     """The careful lab's chain on one lot at settings pv: Utah unless the lot already passed the
207:     density gate, then Mines, microscopy, tensile, the screen, and immersion if the screen passes
208:     (top quartile of screened builds, or with `w_ref` at least the reference coupon's screen index).
209:     Appends the lab's own record (x, y) to rows, with a proxy value for a stopped chain. Returns the
210:     action label."""
211:     L = d.lots[lot]
212:     if need_utah:
213:         # the small step scans travel speed at Utah's full (capped) power, one setting per visit
214:         pu = [P_CAP, UTAH_SCAN[len(L["utah"]) % len(UTAH_SCAN)]]
215:         r = lab.utah(lot, pu)
216:         if r < RHO_GATE:
217:             xu = torch.cat([L["c"], torch.tensor([L["a"]] + pu)])
218:             rows.append((xu, -LAMBDA * float(burden(xu))))  # did not deposit dense: recorded as a failure
219:             return "utah-fail"
220:     if not any(B["lot"] == lot for B in d.builds):
221:         pv = list(max(L["utah"], key=lambda t: t[2])[:2])  # the first Mines build on a lot uses the setting that passed at Utah
222:     x = torch.cat([L["c"], torch.tensor([L["a"]] + [float(v) for v in pv])])
223:     b = lab.mines(lot, pv)
224:     lab.microscopy(b)
225:     U = lab.tensile(b)
226:     w = lab.screen(b)
227:     gate = w_gate(d) if w_ref is None else w_ref
228:     if w >= gate:
229:         lab.immersion(b)
230:         rows.append((x, d.builds[b]["J"]))
231:         return "full"
232:     rows.append((x, U * float(sig(torch.tensor((w - gate) / 0.05))) - LAMBDA * float(burden(x))))  # the screen stands in for immersion
233:     return "screen-fail"

236: def run_funnel_stages(lab, d, comps, fom, fvals, rows):
237:     """A fixed top-k funnel at every gate (CALPHAD, powder, Utah, screen); returns the finalist
238:     compositions and the record of what each gate kept and rejected."""
239:     k1, k2, k3, k4 = FUNNEL_K
240:     gates = []
241:     order = fom.argsort(descending=True).tolist()
242:     keep = [i for i in order if float(fom[i]) > -9.0][:k1]
243:     gates.append({"gate": "CALPHAD", "rule": f"top {k1} by s - 0.3 B with f <= {F_SAFE}", "kept": [comps[i].tolist() for i in keep],
244:                   "rejected": [comps[i].tolist() for i in order if i not in keep]})
245:     lots = [lab.atomize(comps[i], torch.tensor(0.5)) for i in keep]
246:     qs = sorted(lots, key=lambda l: -d.lots[l]["q"])
247:     keep2 = qs[:k2]
248:     gates.append({"gate": "powder", "rule": f"top {k2} of {len(lots)} lots by q (atomized at a = 0.5)", "kept": [d.lots[l]["c"].tolist() for l in keep2],
249:                   "rejected": [d.lots[l]["c"].tolist() for l in qs[k2:]]})
250:     best = {}
251:     for l in keep2:
252:         best[l] = max((lab.utah(l, [0.5, v]), v) for v in (0.25, 0.45, 0.65))
253:     us = sorted(keep2, key=lambda l: -best[l][0])
254:     keep3 = us[:k3]
255:     gates.append({"gate": "Utah", "rule": f"top {k3} of {len(keep2)} lots by the best of three small-DED settings", "kept": [d.lots[l]["c"].tolist() for l in keep3],
256:                   "rejected": [d.lots[l]["c"].tolist() for l in us[k3:]]})
257:     builds = []
258:     for l in keep3:
259:         v = best[l][1]
260:         for pv in ([0.5, v], [0.75, min(v + 0.25, 1.0)]):
261:             b = lab.mines(l, pv)
262:             lab.microscopy(b)
263:             lab.tensile(b)
264:             lab.screen(b)
265:             builds.append(b)
266:     ws = sorted(builds, key=lambda b: -d.builds[b]["w"])
267:     keep4 = ws[:k4]
268:     for b in builds:
269:         x = design_of_build(d, b)
270:         B = d.builds[b]
271:         if b in keep4:
272:             lab.immersion(b)
273:             rows.append((x, B["J"]))
274:         else:
275:             rows.append((x, B["U"] * float(sig(torch.tensor((B["w"] - d.builds[keep4[-1]]["w"]) / 0.05))) - LAMBDA * float(burden(x))))
276:     gates.append({"gate": "screen", "rule": f"top {k4} of {len(builds)} Mines builds by the one-hour screen go to immersion",
277:                   "kept": [design_of_build(d, b).tolist() for b in keep4], "rejected": [design_of_build(d, b).tolist() for b in ws[k4:]]})
278:     return torch.stack([d.lots[l]["c"] for l in keep3]), gates

281: def run_full_policy(kind, truth, seed, budget):
282:     d, lab = initial_data(truth, seed)
283:     rec = Recorder(truth, d)
284:     net = Network(d)
285:     lab.net = net
286:     rec(net, 0, "init")
287:     g = torch.Generator().manual_seed(3000 + seed)
288:     it = 0
289:     rows = list(d.full)  # the lab's own records for tiered screening and the funnel
290:     gates = None
291:     w_ref = w_reference(truth) if kind == "tiered_ni_ref" else None
292:     if kind in ("tiered", "tiered_ni", "tiered_ni_ref", "funnel"):
293:         shortlist, comps, fom, fvals = calphad_shortlist(d, lab, seed, ni_floor=NI_FLOOR if kind in ("tiered_ni", "tiered_ni_ref") else None)
294:         net = Network(d)
295:         lab.net = net
296:         it += 1
297:         rec(net, it, "C-screen")
298:         if kind == "funnel":
299:             shortlist, gates = run_funnel_stages(lab, d, comps, fom, fvals, rows)
300:             net = Network(d)
301:             lab.net = net
302:             it += 1
303:             rec(net, it, "funnel")
304:     reuse = kind in ("tiered", "tiered_ni", "tiered_ni_ref", "funnel", "eifn_reuse")
305:     cheapest = COSTS["M"] + TAIL if reuse else FULL_COST
306:     while d.spent + cheapest <= budget:
307:         it += 1
308:         left = budget - d.spent
309:         X = torch.stack([x for x, _ in d.full])
310:         y = torch.tensor([v for _, v in d.full])
311:         if kind == "eifn_reuse":
312:             # ablation: network EI per dollar over new designs and new Mines sessions on
313:             # existing lots (partial evaluation without KG's stage choice; every remaining stage runs)
314:             best = float(y.max())
315:             Pn = torch.cat([sobol(1024, D, seed=it + 50 * seed), local_perturb(X, 16, 0.05, g)])
316:             pools, lots = [Pn], [-1] * Pn.shape[0]
317:             costs = [torch.full((Pn.shape[0],), FULL_COST)]
318:             for i, L in enumerate(d.lots):
319:                 pv = sobol(64, 2, seed=it + 50 * seed + i)
320:                 pools.append(torch.cat([L["c"].expand(64, NC), torch.full((64, 1), L["a"]), pv], -1))
321:                 costs.append(torch.full((64,), lab.mines_cost(i) + TAIL))
322:                 lots += [i] * 64
323:             P, cost = torch.cat(pools), torch.cat(costs)
324:             ei, _ = network_ei_pool(net, d, P, eps_for(P.shape[0], 64, it), best)
325:             score = ei / cost
326:             score[cost > left] = -1.0
327:             j = int(score.argmax())
328:             if float(score[j]) < 0:
329:                 break
330:             if lots[j] >= 0:
331:                 b = lab.mines(lots[j], P[j, IPV])
332:                 lab.microscopy(b)
333:                 lab.tensile(b)
334:                 lab.screen(b)
335:                 lab.immersion(b)
336:             else:
337:                 lab.full_eval(P[j])
338:             net = Network(d)
339:             lab.net = net
340:             rec(net, it, "reuse" if lots[j] >= 0 else "full")
341:             continue
342:         if reuse:
343:             # Stage 2, done the way a careful lab would: EI on its own records over the
344:             # shortlist only, reusing each lot (and its scale-up) across sessions.
345:             gp = lab_gp(rows)
346:             pools, costs, lots, needs = [], [], [], []
347:             for c in shortlist:
348:                 for i, L in enumerate(d.lots):
349:                     if torch.allclose(L["c"], c):
350:                         if not lot_passed_utah(L) and len(L["utah"]) >= UTAH_TRIES:
351:                             continue  # retired: this powder does not deposit dense at Utah
352:                         pv = sobol(64, 2, seed=it + 50 * seed + i)
353:                         pools.append(torch.cat([c.expand(64, NC), torch.full((64, 1), L["a"]), pv], -1))
354:                         passed = lot_passed_utah(L)
355:                         costs.append(torch.full((64,), (0.0 if passed else COSTS["U"]) + lab.mines_cost(i) + TAIL))
356:                         lots += [i] * 64
357:                         needs += [not passed] * 64
358:                 if kind in ("tiered", "tiered_ni", "tiered_ni_ref"):  # a new lot of a shortlisted composition
359:                     apv = sobol(128, 3, seed=it + 50 * seed)
360:                     pools.append(torch.cat([c.expand(128, NC), apv], -1))
361:                     costs.append(torch.full((128,), COSTS["A"] + COSTS["U"] + M_FIRST + TAIL))
362:                     lots += [-1] * 128
363:                     needs += [True] * 128
364:             if not pools:
365:                 break
366:             P, cost = torch.cat(pools), torch.cat(costs)
367:             score = expected_improvement(gp, P, max(v for _, v in rows))
368:             score[cost > left] = -1.0
369:             j = int(score.argmax())
370:             if float(score[j]) < 0:
371:                 break
372:             if lots[j] >= 0:
373:                 action = chain(lab, d, lots[j], P[j, IPV], needs[j], rows, w_ref)
374:             else:
375:                 lot = lab.atomize(P[j, COMP], P[j, IA])
376:                 action = chain(lab, d, lot, P[j, IPV], True, rows, w_ref)
377:             net = Network(d)
378:             lab.net = net
379:             rec(net, it, action)
380:             continue
381:         if kind == "random":
382:             x = torch.rand(D, generator=g)
383:         elif kind == "ei":
384:             gp = TinyGP(X, y)
385:             P = torch.cat([sobol(2048, D, seed=it + 50 * seed), local_perturb(X[y.topk(min(5, len(y))).indices], 64, 0.05, g)])
386:             x = P[expected_improvement(gp, P, y.max()).argmax()]
387:         elif kind == "eifn":
388:             best = float(y.max())
389:             P = torch.cat([sobol(1024, D, seed=it + 50 * seed), local_perturb(X, 16, 0.05, g)])
390:             ei, _ = network_ei_pool(net, d, P, eps_for(P.shape[0], 64, it), best)
391:             Q = local_perturb(P[ei.topk(8).indices], 32, 0.03, g)
392:             eq, _ = network_ei_pool(net, d, Q, eps_for(Q.shape[0], 64, it), best)
393:             P, ei = torch.cat([P, Q]), torch.cat([ei, eq])
394:             x = P[ei.argmax()]
395:         else:
396:             raise KeyError(kind)
397:         lab.full_eval(x)
398:         net = Network(d)
399:         lab.net = net
400:         rec(net, it, "full")
401:     return rec.rows, d, {"gates": gates, "shortlist": shortlist.tolist() if kind in ("tiered", "tiered_ni", "tiered_ni_ref", "funnel") else None, "w_ref": w_ref}

405: def run_pkgfn(truth, seed, budget, verbose=False, utah=True, screens=False, audit=False):
406:     """pKGFN over the eight stages. `utah=False` removes the small-DED stage (`mines_only`).
407:     `screens=True` makes it trust the screens (no atomizing a composition the CALPHAD crack
408:     screen rejects, no Mines session on a lot that has not passed the Utah density gate, no
409:     immersion on a build the one-hour screen stops); `audit=True` lets about one in ten stopped
410:     candidates go on anyway, plus those just either side of each cutoff (Advance 5)."""
411:     d, lab = initial_data(truth, seed)
412:     rec = Recorder(truth, d)
413:     net = Network(d)
414:     lab.net = net
415:     rec(net, 0, "init")
416:     g = torch.Generator().manual_seed(5000 + seed)
417:     gh_t, gh_w = gauss_hermite(10)
418:     w1 = gh_t.unsqueeze(-1)
419:     # 4-output CALPHAD fantasy: 2-point Gauss-Hermite product rule
420:     pm = torch.tensor([-1.0, 1.0])
