53: def wilcoxon_row(A, B, ystar_a, ystar_b, spends=(25_000, 50_000, 100_000)):
54:     from scipy.stats import wilcoxon
55: 
56:     a = {r["seed"]: curve(r["rows"]) / ystar_a for r in A}
57:     b = {r["seed"]: curve(r["rows"]) / ystar_b for r in B}
58:     seeds = sorted(set(a) & set(b))
59:     cells = []
60:     for s_ in spends:
61:         i = int(np.searchsorted(GRID, s_))
62:         d = np.array([a[s][i] - b[s][i] for s in seeds])
63:         p = wilcoxon(d).pvalue if np.any(d != 0) else 1.0
64:         cells.append(f"{d.mean() * 100:+.0f} points ({(d > 0).sum()}/{(d == 0).sum()}/{(d < 0).sum()}; p = {p:.2g})")
65:     return len(seeds), cells

171: def first_positive_table():
172:     """How long each policy takes to demonstrate its first alloy with J > 0 (it must pass immersion and
173:     out-earn its burden), and how often pKGFN's KG was flat before and after that point."""
174:     cu = load_dir(CU / "results" / "k1e1")
175:     light = load_dir(CU / "quality" / "results-light" / "k1e1")
176:     rows = ["| Policy | n | seeds whose shared seed data already holds an alloy with J > 0 | median new spend to the first alloy with J > 0 (seeds that reach it) | KG-flat fallback share before it | after it |", "|---|---|---|---|---|---|"]
177:     sets = [(k, cu[k]) for k in ("pkgfn", "eifn_reuse", "eifn", "ei", "random", "tiered", "tiered_ni", "tiered_ni_ref") if k in cu]
178:     if "pkgfn" in light:
179:         sets.insert(1, ("pkgfn (light settings)", light["pkgfn"]))
180:     for k, runs in sets:
181:         start = sum(1 for r in runs if r["rows"][0]["best_validated"] > 0)
182:         sp, fb_b, fb_a = [], [0, 0], [0, 0]
183:         for r in runs:
184:             hit = next((x["spent"] for x in r["rows"] if x["best_validated"] > 0), None)
185:             sp.append(hit if hit is not None else np.inf)
186:             for x in r["rows"][1:]:
187:                 if "fallback" not in x:
188:                     continue
189:                 tgt = fb_b if x["best_validated"] <= 0 else fb_a
190:                 tgt[0] += int(x["fallback"])
191:                 tgt[1] += 1
192:         sp = np.array(sp)
193:         fb = (f"{fb_b[0] / fb_b[1]:.0%} of {fb_b[1]}" if fb_b[1] else "", f"{fb_a[0] / fb_a[1]:.0%} of {fb_a[1]}" if fb_a[1] else "")
194:         name = SHORT.get(k, k)
195:         rows.append(f"| {name} | {len(runs)} | {start} | {fmt_cost(np.median(sp))} ({np.isfinite(sp).sum()} of {len(sp)}) | {fb[0]} | {fb[1]} |")
196:     return "\n".join(rows)
